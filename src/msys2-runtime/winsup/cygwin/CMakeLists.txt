# CMake build for the Cygwin/MSYS runtime DLL (winsup/cygwin).
#
# This is a replacement for the autotools build (Makefile.am) using
# CMake + Ninja + Clang.  All build products referenced from the
# pre-existing autotools target tree are newlib artifacts only.
#
# Configure (from a MSYS2 bash with clang/cmake/ninja in PATH):
#   cmake -G Ninja \
#         -S winsup/cygwin -B ../../build/cmake-clang \
#         -DCMAKE_TOOLCHAIN_FILE=$PWD/cmake/toolchain-clang-cygwin.cmake
#   cmake --build ../../build/cmake-clang

cmake_minimum_required(VERSION 3.20)
project(cygwin LANGUAGES C CXX ASM)

get_filename_component(CS "${CMAKE_CURRENT_SOURCE_DIR}" ABSOLUTE)
get_filename_component(RUNTIME_SRC "${CS}/../.." ABSOLUTE)
set(GEN "${CMAKE_CURRENT_BINARY_DIR}")

# ---------------------------------------------------------------------------
# Inputs that are not part of this CMake project
# ---------------------------------------------------------------------------
# Reuse the newlib build tree produced by the autotools bootstrap.  Only
# newlib libraries are taken from there; every winsup source is rebuilt here.
set(TARGET_BUILD_DIR "${RUNTIME_SRC}/x86_64-pc-cygwin" CACHE PATH
    "Target build dir holding newlib/targ-include and newlib libs")
set(NEWLIB_LIBC    "${TARGET_BUILD_DIR}/newlib/libc.a" CACHE FILEPATH "")
set(NEWLIB_LIBM    "${TARGET_BUILD_DIR}/newlib/libm.a" CACHE FILEPATH "")
set(CYGSERVER_LIB  "${TARGET_BUILD_DIR}/winsup/cygserver/libcygserver.a" CACHE FILEPATH "")

option(CYGWIN_DEBUGGING "Build DLL with DEBUGGING consistency checks" OFF)
set(CYGWIN_OPTIMIZE "-O2;-g;-gdwarf-4;-fno-integrated-as")

find_program(BASH_EXECUTABLE   bash    REQUIRED)
find_program(PERL_EXECUTABLE   perl    REQUIRED)
find_program(PATCH_EXECUTABLE  patch   REQUIRED)
find_program(WINDRES_EXECUTABLE windres HINTS "${MSYS2_ROOT}/usr/bin" REQUIRED)
find_program(GIT_EXECUTABLE    git)

# ---------------------------------------------------------------------------
# Version information (configure.ac substitutes these)

set(MSYS2_RUNTIME_COMMIT_SHORT "")
set(MSYS2_RUNTIME_COMMIT_HEX "0")
if(GIT_EXECUTABLE)
  execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
                  WORKING_DIRECTORY "${RUNTIME_SRC}"
                  OUTPUT_VARIABLE _commit OUTPUT_STRIP_TRAILING_WHITESPACE
                  RESULT_VARIABLE _rc ERROR_QUIET)
  if(_rc EQUAL 0 AND _commit MATCHES "^[0-9a-f]+$")
    set(MSYS2_RUNTIME_COMMIT "${_commit}")
    string(SUBSTRING "${_commit}" 0 8 MSYS2_RUNTIME_COMMIT_SHORT)
    set(MSYS2_RUNTIME_COMMIT_HEX "0x${MSYS2_RUNTIME_COMMIT_SHORT}ul")
  endif()
endif()

string(REPLACE ";" " " CYGWIN_TARGET_FLAGS_STR "${CYGWIN_TARGET_FLAGS}")
# clang 22 emits LLVM-only pseudo-ops and COFF section forms that GNU as
# rejects when -fno-integrated-as is in effect (.seh_startepilogue,
# 4-argument COFF .section, .p2align fill byte).  Build the filtering
# assembler wrapper once and let clang find it via -B.
set(AS_FILTER_DIR "${GEN}/asbin")
file(MAKE_DIRECTORY "${AS_FILTER_DIR}")
execute_process(
  COMMAND "${MSYS2_ROOT}/mingw64/bin/clang.exe" -O2
          -o "${AS_FILTER_DIR}/as.exe" "${CS}/cmake/tools/filter_as.c"
  RESULT_VARIABLE _as_filter_rc)
if(NOT _as_filter_rc EQUAL 0)
  message(FATAL_ERROR "failed to build the assembler filter")
endif()

    "-DMSYS2_RUNTIME_COMMIT_HEX=${MSYS2_RUNTIME_COMMIT_HEX}"

# ---------------------------------------------------------------------------
# new-msys-2.0.dll
# ---------------------------------------------------------------------------
# Cygwin's GCC link driver injects --wrap=_Znwm etc. by default, which
# redirects the DLL's own operator new to the libstdc++ wrappers (they read
# user_data->cxx_malloc, not ready during dll_crt0_0).  Autotools suppresses
# them with -mno-use-libstdc-wrappers; clang delegates linking to that driver
# and rejects the flag itself, so link through the driver directly.
set(CYGWIN_LINK_DRIVER "${MSYS2_ROOT}/usr/bin/x86_64-pc-cygwin-g++.exe")
set(CYGWIN_LINK_FLAGS
    -mno-use-libstdc-wrappers
    -Wl,--gc-sections
    -nostdlib
    -Wl,--dynamicbase
    -static
    -Wl,--heap=0)

# The object list is far too long for a Windows command line, so feed it
# through a response file.
file(GENERATE OUTPUT "${GEN}/objects.rsp"
     CONTENT "$<JOIN:$<TARGET_OBJECTS:cygdll>,\n>\n")

add_custom_command(OUTPUT "${GEN}/new-msys-2.0.dll" "${GEN}/msysdll.a"
  COMMAND "${CYGWIN_LINK_DRIVER}" ${CYGWIN_LINK_FLAGS}
          "-Wl,-T${GEN}/cygwin.sc"
          "-Wl,--out-implib,${GEN}/msysdll.a"
          -shared -o "${GEN}/new-msys-2.0.dll"
          -e dll_entry "${GEN}/msys.def"
          "@${GEN}/objects.rsp"
          "${GEN}/winver.o"
          "${CYGSERVER_LIB}" "${NEWLIB_LIBM}" "${NEWLIB_LIBC}"
          -lgcc -lkernel32 -lntdll
          "-Wl,-Map,${GEN}/msys.map"
  DEPENDS $<TARGET_OBJECTS:cygdll> cygdll
          "${GEN}/winver.o" "${GEN}/msys.def" "${GEN}/cygwin.sc"
          "${CYGSERVER_LIB}" "${NEWLIB_LIBM}" "${NEWLIB_LIBC}"
  COMMAND_EXPAND_LISTS
  COMMENT "Linking new-msys-2.0.dll")

add_custom_target(msysdll ALL DEPENDS "${GEN}/new-msys-2.0.dll")

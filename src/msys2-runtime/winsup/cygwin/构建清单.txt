# CMake build for the Cygwin/MSYS runtime DLL (winsup/cygwin).
#
# This is a replacement for the autotools build (Makefile.am) using
# CMake + Ninja + Clang.  All build products referenced from the
# pre-existing autotools target tree are newlib artifacts only.
#
# Configure (from a MSYS2 bash with clang/cmake/ninja in PATH):
#   配置 -G Ninja \
#         -S winsup/cygwin -B ../../构建/cmake-clang \
#         -DCMAKE_TOOLCHAIN_FILE=$PWD/配置/工具链-clang-cygwin.cmake
#   构建 -C ../../构建/cmake-clang

cmake_minimum_required(VERSION 3.20)
project(cygwin LANGUAGES C CXX ASM)

# clang 22 emits LLVM-only pseudo-ops and COFF section forms that GNU as
# rejects when -fno-integrated-as is in effect (.seh_startepilogue,
# 4-argument COFF .section, .p2align fill byte).  Build the filtering
# assembler wrapper once and let clang find it via -B.
set(AS_FILTER_DIR "${GEN}/asbin")
file(MAKE_DIRECTORY "${AS_FILTER_DIR}")
execute_process(
  COMMAND "${MSYS2_ROOT}/mingw64/bin/clang.exe" -O2
          -municode -o as.exe "${CS}/配置/tools/filter_as.c"
  WORKING_DIRECTORY "${AS_FILTER_DIR}"
  RESULT_VARIABLE _as_filter_rc)
if(NOT _as_filter_rc EQUAL 0)
  message(FATAL_ERROR "failed to build the assembler filter")
endif()

# ---------------------------------------------------------------------------
# DLL 产物（文件名由 CYGWIN_CHINESE_NAMES 宏控制）
# ---------------------------------------------------------------------------
# Cygwin's GCC link driver injects --wrap=_Znwm etc. by default, which
# redirects the DLL's own operator new to the libstdc++ wrappers (they read
# user_data->cxx_malloc, not ready during dll_crt0_0).  Autotools suppresses
# them with -mno-use-libstdc-wrappers; clang delegates linking to that driver
# and rejects the flag itself, so link through the driver directly.
set(CYGWIN_LINK_DRIVER "${MSYS2_ROOT}/usr/bin/x86_64-pc-cygwin-g++.exe")
set(CYGWIN_LINK_FLAGS
    -mno-use-libstdc-wrappers
    -Wl,--gc-sections
    -nostdlib
    -Wl,--dynamicbase
    -static
    -Wl,--heap=0)

# The object list is far too long for a Windows command line, so feed it
# through a response file.
file(GENERATE OUTPUT "${GEN}/objects.rsp"
     CONTENT "$<JOIN:$<TARGET_OBJECTS:cygdll>,\n>\n")

add_custom_command(OUTPUT "${GEN}/${MSYS_DLL_FILE}" "${GEN}/${MSYS_IMPLIB_FILE}"
  COMMAND "${CYGWIN_LINK_DRIVER}" ${CYGWIN_LINK_FLAGS}
          "-Wl,-T${GEN}/cygwin.sc"
          "-Wl,--out-implib,${GEN}/${MSYS_IMPLIB_FILE}"
          -shared -o "${GEN}/${MSYS_DLL_FILE}"
          -e dll_entry "${GEN}/msys.def"
          "@${GEN}/objects.rsp"
          "${GEN}/winver.o"
          "${CYGSERVER_LIB}" "${NEWLIB_LIBM}" "${NEWLIB_LIBC}"
          -lgcc -lkernel32 -lntdll
          "-Wl,-Map,${GEN}/${MSYS_MAP_FILE}"
  DEPENDS $<TARGET_OBJECTS:cygdll> cygdll
          "${GEN}/winver.o" "${GEN}/msys.def" "${GEN}/cygwin.sc"
          "${CYGSERVER_LIB}" "${NEWLIB_LIBM}" "${NEWLIB_LIBC}"
  COMMAND_EXPAND_LISTS
  COMMENT "Linking ${MSYS_DLL_FILE}")

add_custom_target(msysdll ALL DEPENDS "${GEN}/${MSYS_DLL_FILE}")

      -DNTDDI_VERSION=WDK_NTDDI_VERSION
      -Wimplicit-fallthrough=4
      -Wwrite-strings)
  set(UTILS_CXX_FLAGS
      ${UTILS_C_FLAGS}
      -fno-rtti -fno-exceptions -fno-use-cxa-atexit)

  set(UTILS_INC
      "-I${CS}/local_includes"
      "-I${GEN}"
      "-I${UTILS_SRC}"
      -isystem ${CS}/include
      -isystem ${TARGET_BUILD_DIR}/newlib/targ-include
      -isystem ${RUNTIME_SRC}/newlib/libc/include)

  set(UTILS_LINK_DRIVER "${MSYS2_ROOT}/usr/bin/x86_64-pc-cygwin-g++.exe")
  set(UTILS_LD_FLAGS -static -Wl,--enable-auto-import)

  set(UTILS_OUTPUTS "")
  function(cygwin_utils_tool name)
    cmake_parse_arguments(T "" "" "SOURCES;CXXFLAGS;LIBS;LDFLAGS" ${ARGN})
    if(CYGWIN_CHINESE_NAMES)
      set(_outname "${UTILS_CN_NAME_${name}}")
    else()
      set(_outname "${name}")
    endif()
    if(NOT _outname)
      message(FATAL_ERROR "缺少 ${name} 的中文产物名（UTILS_CN_NAME_${name}）")
    endif()
    file(MAKE_DIRECTORY "${UTILS_OUT}/obj/${name}")
    set(_objs "")
    foreach(src IN LISTS T_SOURCES)
      get_filename_component(_base "${src}" NAME_WE)
      get_filename_component(_ext "${src}" EXT)
      set(_obj "${UTILS_OUT}/obj/${name}/${_base}.o")
      if(_ext STREQUAL ".cc")
        set(_cc "${CMAKE_CXX_COMPILER}")
        set(_flags ${UTILS_CXX_FLAGS} ${T_CXXFLAGS})
      else()
        set(_cc "${CMAKE_C_COMPILER}")
        set(_flags ${UTILS_C_FLAGS})
      endif()
      add_custom_command(OUTPUT "${_obj}"
        COMMAND "${_cc}" ${_flags} ${UTILS_INC}
                -c "${UTILS_SRC}/${src}" -o "${_obj}"
        DEPENDS "${UTILS_SRC}/${src}" "${GEN}/config.h"
        COMMAND_EXPAND_LISTS
        COMMENT "编译 ${name}/${src}")
      list(APPEND _objs "${_obj}")
    endforeach()
    if(T_LDFLAGS)
      set(_ldflags ${T_LDFLAGS})
    else()
      set(_ldflags ${UTILS_LD_FLAGS})
    endif()
    add_custom_command(OUTPUT "${UTILS_OUT}/${_outname}.exe"
      COMMAND "${UTILS_LINK_DRIVER}" ${_ldflags} -o "${UTILS_OUT}/${_outname}.exe"
              ${_objs} "${GEN}/${MSYS_IMPLIB_FILE}" -lnetapi32 ${T_LIBS}
      DEPENDS ${_objs} "${GEN}/${MSYS_IMPLIB_FILE}" "${GEN}/${MSYS_DLL_FILE}"
      COMMENT "链接 ${_outname}.exe")
    set(UTILS_OUTPUTS "${UTILS_OUTPUTS}" "${UTILS_OUT}/${_outname}.exe" PARENT_SCOPE)
  endfunction()

  cygwin_utils_tool(chattr        SOURCES chattr.c)

  cygwin_utils_tool(mkpasswd      SOURCES mkpasswd.c)
  cygwin_utils_tool(mount         SOURCES mount.cc path.cc CXXFLAGS -DFSTAB_ONLY)
  cygwin_utils_tool(newgrp        SOURCES newgrp.c    LIBS -luserenv)
